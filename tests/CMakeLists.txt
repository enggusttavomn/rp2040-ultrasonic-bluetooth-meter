cmake_minimum_required(VERSION 3.13)
project(hcsr04_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(test_hcsr04 test_hcsr04.c ../src/hcsr04.c)
target_include_directories(test_hcsr04 PRIVATE stubs ../inc)
if(MSVC)
    target_compile_options(test_hcsr04 PRIVATE /W4)
else()
    target_compile_options(test_hcsr04 PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME hcsr04 COMMAND test_hcsr04)
